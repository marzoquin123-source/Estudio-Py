@ 2-1
def cuadrado_de_texto(texto):

    return numero ** 2

@ 2-2
def promedio_de_textos(a, b):

    y = float(b)

@ 3-1
def resumen_dias(dias):
    semanas = dias // 7
    sobrantes = dias % 7
    return str(semanas) + " semanas y " + str(sobrantes) + " dias"

@ 3-2
def cuota_por_persona(total, personas):
    total /= personas
    return total

@ 3-3
def saldo_final(saldo, deposito, retiro):
    saldo += deposito
    saldo -= retiro
    return saldo

@ 11-3
def es_bisiesto(anio):
    if anio % 400 == 0:
        return True
    elif anio % 100 == 0:
        return False
    elif anio % 4 == 0:
        return True
    else:
        return False

@ 12-1
def puntos_del_numero(n):
    puntos = 0
    if n > 0:
        puntos += 1
    if n % 2 == 0:
        puntos += 2
    if n > 100:
        puntos += 4
    return puntos

@ 12-2

    elif t >= 20:
        return "agradable"
    elif t >= 10:
        return "fresco"
    else:
        return "frio"

@ 12-3
def fizzbuzz(n):
    if n % 15 == 0:
        return "FizzBuzz"
    elif n % 3 == 0:
        return "Fizz"
    elif n % 5 == 0:
        return "Buzz"
    else:
        return str(n)

@ 13-1
def suma_hasta(n):
    i = 1
    total = 0
    while i <= n:
        total += i
        i += 1
    return total

@ 13-2

@ 14-1
def suma_de_digitos(n):

@ 14-2
def invertir_numero(n):
    r = 0

@ 14-3
def meses_para_meta(ahorro, aporte, meta):
    meses = 0
    while ahorro < meta:
        ahorro += aporte
        meses += 1
    return meses

@ 15-1
def suma_pares_hasta(n):
    total = 0
    for i in range(1, n + 1):
        if i % 2 == 0:
            total += i
    return total

@ 15-2

@ 16-1
def contar_vocales(texto):
    texto = texto.lower()
    total = 0
    for letra in texto:
        if letra == "a" or letra == "e" or letra == "i" or letra == "o" or letra == "u":
            total += 1
    return total

@ 16-2
def invertir_texto(texto):

@ 16-3
def solo_digitos(texto):

@ 19-2
def mayor_por_posicion(a, b, c):
    resultado = ()
    for i in range(4):
        mayor = a[i]
        if b[i] > mayor:
            mayor = b[i]
        if c[i] > mayor:
            mayor = c[i]
        resultado = resultado + (mayor,)
    return resultado

@ 19-3
def promedios_mayores_a_cinco(a, b, c):
    resultado = ()
    for i in range(4):
        promedio = (a[i] + b[i] + c[i]) / 3
        if promedio > 5:
            resultado = resultado + (promedio,)
    return resultado

@ 20-1
def sumas_sin_repetir(a, b, c):
    resultado = set()
    for i in range(4):
        resultado.add(a[i] + b[i] + c[i])
    return resultado

@ 20-2
def restas_en_texto(a, b, c):
    resultado = []
    for i in range(4):
        resultado = resultado + [str(a[i] - b[i] - c[i])]
    return ",".join(resultado)

@ 20-3
def nombres_completos(a, b, c):
    resultado = ()
    for i in range(4):
        frase = a[i].strip() + " " + b[i].strip() + " " + c[i].strip()
        resultado = resultado + (frase.title(),)
    return resultado
